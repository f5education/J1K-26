# --- Web Framework & ASGI Server ---
# Core async web framework serving /healthz, /api/client-config, POST /api/proxy,

# Production ASGI HTTP server that binds 0.0.0.0:PORT inside Docker or local venvs.
uvicorn==0.32.1

# --- Outbound HTTP Client ---
